import pytest
import numpy as np
import tensorflow as tf
from keras.backend.tensorflow_backend import elu

def test_positive_input_identity():
    """Verify that for any input x > 0, the output is exactly x. This tests the positive branch of the ELU definition."""
    x_val = np.array([1.0, 2.0, 3.0], dtype=np.float32)
    x = tf.constant(x_val)
    output = elu(x, alpha=1.0)
    
    with tf.Session() as sess:
        result = sess.run(output)
        np.testing.assert_allclose(result, x_val, rtol=1e-5)

def test_negative_input_formula():
    """Verify that for any input x < 0, the output equals alpha * (exp(x) - 1). Use a standard alpha of 1.0 for this specific check to isolate the exponential logic."""
    x_val = np.array([-1.0, -2.0, -3.0], dtype=np.float32)
    expected = 1.0 * (np.exp(x_val) - 1.0)
    x = tf.constant(x_val)
    output = elu(x, alpha=1.0)
    
    with tf.Session() as sess:
        result = sess.run(output)
        np.testing.assert_allclose(result, expected, rtol=1e-5)

def test_zero_input_returns_zero():
    """Verify that when input x is 0, the output is 0. This checks the boundary condition where both branches theoretically converge."""
    x_val = np.array([0.0], dtype=np.float32)
    x = tf.constant(x_val)
    output = elu(x, alpha=1.0)
    
    with tf.Session() as sess:
        result = sess.run(output)
        np.testing.assert_allclose(result, [0.0], rtol=1e-5)

def test_mixed_input_element_wise():
    """Verify that the function correctly applies the positive branch to positive values and the negative branch to negative values within the same tensor. Input: [-1.0, 0.0, 1.0], Expected: [alpha*(e^-1 - 1), 0.0, 1.0]."""
    x_val = np.array([-1.0, 0.0, 1.0], dtype=np.float32)
    # Expected:
    # -1.0: 1.0 * (exp(-1) - 1)
    # 0.0: 0.0
    # 1.0: 1.0
    expected = np.array([np.exp(-1.0) - 1.0, 0.0, 1.0], dtype=np.float32)
    x = tf.constant(x_val)
    output = elu(x, alpha=1.0)
    
    with tf.Session() as sess:
        result = sess.run(output)
        np.testing.assert_allclose(result, expected, rtol=1e-5)

def test_default_alpha_value():
    """Verify that if alpha is not provided, it defaults to 1.0. The result of elu(x) should be identical to elu(x, alpha=1.0)."""
    x_val = np.array([-1.0, 1.0], dtype=np.float32)
    x = tf.constant(x_val)
    output_default = elu(x)
    output_explicit = elu(x, alpha=1.0)
    
    with tf.Session() as sess:
        result_default = sess.run(output_default)
        result_explicit = sess.run(output_explicit)
        np.testing.assert_allclose(result_default, result_explicit, rtol=1e-5)

def test_custom_alpha_scaling():
    """Verify that changing the alpha parameter scales the negative section correctly. For x < 0, if alpha=2.0, the output should be exactly double that of alpha=1.0."""
    x_val = np.array([-1.0, -2.0], dtype=np.float32)
    x = tf.constant(x_val)
    output_alpha1 = elu(x, alpha=1.0)
    output_alpha2 = elu(x, alpha=2.0)
    
    with tf.Session() as sess:
        res1 = sess.run(output_alpha1)
        res2 = sess.run(output_alpha2)
        # For negative inputs, res2 should be 2 * res1
        np.testing.assert_allclose(res2, 2.0 * res1, rtol=1e-5)

def test_alpha_zero_behavior():
    """Verify that when alpha=0.0, the negative section becomes 0.0 for all x <= 0. This effectively mimics ReLU behavior for negative inputs."""
    x_val = np.array([-1.0, -2.0, 0.0, 1.0], dtype=np.float32)
    x = tf.constant(x_val)
    output = elu(x, alpha=0.0)
    
    with tf.Session() as sess:
        result = sess.run(output)
        # For x <= 0, output should be 0. For x > 0, output should be x.
        expected = np.array([0.0, 0.0, 0.0, 1.0], dtype=np.float32)
        np.testing.assert_allclose(result, expected, rtol=1e-5)

def test_large_negative_stability():
    """Verify numerical stability for large negative inputs. As x -> -infinity, exp(x) -> 0, so output should approach -alpha. Input: -100.0, alpha=1.0, Expected: approx -1.0."""

   142	def get_session():
   143	    """Returns the TF session to be used by the backend.
   144	
   145	    If a default TensorFlow session is available, we will return it.
   146	
   147	    Else, we will return the global Keras session.
   148	
   149	    If no global Keras session exists at this point:
   150	    we will create a new global session.
   151	
   152	    Note that you can manually set the global session
   153	    via `K.set_session(sess)`.
   154	
   155	    # Returns
   156	        A TensorFlow session.
   157	    """
   158	    global _SESSION
   159	    if tf.get_default_session() is not None:
   160	        session = tf.get_default_session()
   161	    else:
   162	        if _SESSION is None:
   163	            if not os.environ.get('OMP_NUM_THREADS'):
   164	                config = tf.ConfigProto(allow_soft_placement=True)
   165	            else:
   166	                num_thread = int(os.environ.get('OMP_NUM_THREADS'))
   167	                config = tf.ConfigProto(intra_op_parallelism_threads=num_thread,
   168	                                        allow_soft_placement=True)
   169	            _SESSION = tf.Session(config=config)
   170	        session = _SESSION
   171	    if not _MANUAL_VAR_INIT:
   172	        with session.graph.as_default():
   173	            variables = tf.global_variables()
   174	            candidate_vars = []
   175	            for v in variables:
   176	                if not getattr(v, '_keras_initialized', False):
   177	                    candidate_vars.append(v)
   178	            if candidate_vars:
   179	                # This step is expensive, so we only run it on variables
   180	                # not already marked as initialized.
   181	                is_initialized = session.run(
   182	                    [tf.is_variable_initialized(v) for v in candidate_vars])
   183	                uninitialized_vars = []
   184	                for flag, v in zip(is_initialized, candidate_vars):
   185	                    if not flag:
   186	                        uninitialized_vars.append(v)
   187	                    v._keras_initialized = True
   188	                if uninitialized_vars:
   189	                    session.run(tf.variables_initializer(uninitialized_vars))
   190	    # hack for list_devices() function.
   191	    # list_devices() function is not available under tensorflow r1.3.
   192	    if not hasattr(session, 'list_devices'):
   193	        session.list_devices = lambda: device_lib.list_local_devices()
   194	    return session

   431	    # Examples
   432	    ```python
   433	        >>> from keras import backend as K
   434	        >>> from keras.layers import Input, Dense
   435	        >>> np_var = numpy.array([1, 2])
   436	        >>> K.is_keras_tensor(np_var) # A numpy array is not a symbolic tensor.
   437	        ValueError
   438	        >>> k_var = tf.placeholder('float32', shape=(1,1))
   439	        >>> K.is_keras_tensor(k_var) # A variable indirectly created outside of keras is not a Keras tensor.
   440	        False
   441	        >>> keras_var = K.variable(np_var)
   442	        >>> K.is_keras_tensor(keras_var)  # A variable created with the keras backend is not a Keras tensor.
   443	        False
   444	        >>> keras_placeholder = K.placeholder(shape=(2, 4, 5))
   445	        >>> K.is_keras_tensor(keras_placeholder)  # A placeholder is not a Keras tensor.
   446	        False
   447	        >>> keras_input = Input([10])
   448	        >>> K.is_keras_tensor(keras_input) # An Input is a Keras tensor.
   449	        True
   450	        >>> keras_layer_output = Dense(10)(keras_input)
   451	        >>> K.is_keras_tensor(keras_layer_output) # Any Keras layer output is a Keras tensor.
   452	        True
   453	    ```
   454	    """
   455	    if not isinstance(x, (tf.Tensor,
   456	                          tf_variables.Variable,
   457	                          tf.SparseTensor)):
   458	        raise ValueError('Unexpectedly found an instance of type `' + str(type(x)) + '`. '
   459	                         'Expected a symbolic tensor instance.')
   460	    return hasattr(x, '_keras_history')

   464	    """Instantiates a placeholder tensor and returns it.

   999	def dot(x, y):
  1000	    """Multiplies 2 tensors (and/or variables) and returns a *tensor*.
  1001	
  1002	    When attempting to multiply a nD tensor
  1003	    with a nD tensor, it reproduces the Theano behavior.
  1004	    (e.g. `(2, 3) * (4, 3, 5) -> (2, 4, 5)`)
  1005	
  1006	    # Arguments
  1007	        x: Tensor or variable.
  1008	        y: Tensor or variable.
  1009	
  1010	    # Returns
  1011	        A tensor, dot product of `x` and `y`.
  1012	
  1013	    # Examples
  1014	    ```python
  1015	        # dot product between tensors
  1016	        >>> x = K.placeholder(shape=(2, 3))
  1017	        >>> y = K.placeholder(shape=(3, 4))
  1018	        >>> xy = K.dot(x, y)
  1019	        >>> xy
  1020	        <tf.Tensor 'MatMul_9:0' shape=(2, 4) dtype=float32>
  1021	    ```
  1022	
  1023	    ```python
  1024	        # dot product between tensors
  1025	        >>> x = K.placeholder(shape=(32, 28, 3))
  1026	        >>> y = K.placeholder(shape=(3, 4))
  1027	        >>> xy = K.dot(x, y)
  1028	        >>> xy
  1029	        <tf.Tensor 'MatMul_9:0' shape=(32, 28, 4) dtype=float32>
  1030	    ```
  1031	
  1032	    ```python
  1033	        # Theano-like behavior example
  1034	        >>> x = K.random_uniform_variable(shape=(2, 3), low=0, high=1)
  1035	        >>> y = K.ones((4, 3, 5))
  1036	        >>> xy = K.dot(x, y)
  1037	        >>> K.int_shape(xy)
  1038	        (2, 4, 5)
  1039	    ```
  1040	    """
  1041	    if ndim(x) is not None and (ndim(x) > 2 or ndim(y) > 2):
  1042	        x_shape = []
  1043	        for i, s in zip(int_shape(x), tf.unstack(tf.shape(x))):
  1044	            if i is not None:
  1045	                x_shape.append(i)
  1046	            else:
  1047	                x_shape.append(s)
  1048	        x_shape = tuple(x_shape)
  1049	        y_shape = []
  1050	        for i, s in zip(int_shape(y), tf.unstack(tf.shape(y))):
  1051	            if i is not None:
  1052	                y_shape.append(i)
  1053	            else:
  1054	                y_shape.append(s)
  1055	        y_shape = tuple(y_shape)
  1056	        y_permute_dim = list(range(ndim(y)))
  1057	        y_permute_dim = [y_permute_dim.pop(-2)] + y_permute_dim
  1058	        xt = tf.reshape(x, [-1, x_shape[-1]])
  1059	        yt = tf.reshape(tf.transpose(y, perm=y_permute_dim), [y_shape[-2], -1])
  1060	        return tf.reshape(tf.matmul(xt, yt),
  1061	                          x_shape[:-1] + y_shape[:-2] + y_shape[-1:])
  1062	    if is_sparse(x):
  1063	        out = tf.sparse_tensor_dense_matmul(x, y)
  1064	    else:
  1065	        out = tf.matmul(x, y)
  1066	    return out

  1069	def batch_dot(x, y, axes=None):
  1070	    """Batchwise dot product.
  1071	
  1072	    `batch_dot` is used to compute dot product of `x` and `y` when
  1073	    `x` and `y` are data in batch, i.e. in a shape of
  1074	    `(batch_size, :)`.
  1075	    `batch_dot` results in a tensor or variable with less dimensions
  1076	    than the input. If the number of dimensions is reduced to 1,
  1077	    we use `expand_dims` to make sure that ndim is at least 2.
  1078	
  1079	    # Arguments
  1080	        x: Keras tensor or variable with `ndim >= 2`.
  1081	        y: Keras tensor or variable with `ndim >= 2`.
  1082	        axes: list of (or single) int with target dimensions.
  1083	            The lengths of `axes[0]` and `axes[1]` should be the same.
  1084	
  1085	    # Returns
  1086	        A tensor with shape equal to the concatenation of `x`'s shape
  1087	        (less the dimension that was summed over) and `y`'s shape
  1088	        (less the batch dimension and the dimension that was summed over).
  1089	        If the final rank is 1, we reshape it to `(batch_size, 1)`.
  1090	
  1091	    # Examples
  1092	        Assume `x = [[1, 2], [3, 4]]` and `y = [[5, 6], [7, 8]]`
  1093	        `batch_dot(x, y, axes=1) = [[17], [53]]` which is the main diagonal
  1094	        of `x.dot(y.T)`, although we never have to calculate the off-diagonal
  1095	        elements.
  1096	
  1097	        Shape inference:
  1098	        Let `x`'s shape be `(100, 20)` and `y`'s shape be `(100, 30, 20)`.
  1099	        If `axes` is (1, 2), to find the output shape of resultant tensor,
  1100	            loop through each dimension in `x`'s shape and `y`'s shape:
  1101	
  1102	        * `x.shape[0]` : 100 : append to output shape
  1103	        * `x.shape[1]` : 20 : do not append to output shape,
  1104	            dimension 1 of `x` has been summed over. (`dot_axes[0]` = 1)
  1105	        * `y.shape[0]` : 100 : do not append to output shape,
  1106	            always ignore first dimension of `y`
  1107	        * `y.shape[1]` : 30 : append to output shape
  1108	        * `y.shape[2]` : 20 : do not append to output shape,
  1109	            dimension 2 of `y` has been summed over. (`dot_axes[1]` = 2)
  1110	        `output_shape` = `(100, 30)`
  1111	
  1112	    ```python
  1113	        >>> x_batch = K.ones(shape=(32, 20, 1))
  1114	        >>> y_batch = K.ones(shape=(32, 30, 20))
  1115	        >>> xy_batch_dot = K.batch_dot(x_batch, y_batch, axes=[1, 2])
  1116	        >>> K.int_shape(xy_batch_dot)
  1117	        (32, 1, 30)
  1118	    ```
  1119	    """
  1120	    if isinstance(axes, int):
  1121	        axes = (axes, axes)
  1122	    x_ndim = ndim(x)
  1123	    y_ndim = ndim(y)
  1124	    if x_ndim > y_ndim:
  1125	        diff = x_ndim - y_ndim
  1126	        y = tf.reshape(y, tf.concat([tf.shape(y), [1] * (diff)], axis=0))
  1127	    elif y_ndim > x_ndim:
  1128	        diff = y_ndim - x_ndim
  1129	        x = tf.reshape(x, tf.concat([tf.shape(x), [1] * (diff)], axis=0))
  1130	    else:
  1131	        diff = 0
  1132	    if ndim(x) == 2 and ndim(y) == 2:
  1133	        if axes[0] == axes[1]:
  1134	            out = tf.reduce_sum(tf.multiply(x, y), axes[0])
  1135	        else:
  1136	            out = tf.reduce_sum(tf.multiply(tf.transpose(x, [1, 0]), y), axes[1])
  1137	    else:
  1138	        if axes is not None:
  1139	            adj_x = None if axes[0] == ndim(x) - 1 else True
  1140	            adj_y = True if axes[1] == ndim(y) - 1 else None
  1141	        else:
  1142	            adj_x = None
  1143	            adj_y = None
  1144	        out = tf.matmul(x, y, adjoint_a=adj_x, adjoint_b=adj_y)
  1145	    if diff:
  1146	        if x_ndim > y_ndim:
  1147	            idx = x_ndim + y_ndim - 3
  1148	        else:
  1149	            idx = x_ndim - 1
  1150	        out = tf.squeeze(out, list(range(idx, idx + diff)))
  1151	    if ndim(out) == 1:
  1152	        out = expand_dims(out, 1)
  1153	    return out
  1154

  1687	def _regular_normalize_batch_in_training(x, gamma, beta,
  1688	                                         reduction_axes, epsilon=1e-3):
  1689	    """Non-fused version of `normalize_batch_in_training`.
  1690	
  1691	    # Arguments
  1692	        x: Input tensor or variable.
  1693	        gamma: Tensor by which to scale the input.
  1694	        beta: Tensor with which to center the input.
  1695	        reduction_axes: iterable of integers,
  1696	            axes over which to normalize.
  1697	        epsilon: Fuzz factor.
  1698	
  1699	    # Returns
  1700	        A tuple length of 3, `(normalized_tensor, mean, variance)`.
  1701	    """
  1702	    mean, var = tf.nn.moments(x, reduction_axes,
  1703	                              shift=None, name=None, keep_dims=False)
  1704	    normed = tf.nn.batch_normalization(x, mean, var,
  1705	                                       beta, gamma,
  1706	                                       epsilon)
  1707	    return normed, mean, var

  1710	def _broadcast_normalize_batch_in_training(x, gamma, beta,
  1711	                                           reduction_axes, epsilon=1e-3):
  1712	    """Non-fused, broadcast version of `normalize_batch_in_training`.
  1713	
  1714	    # Arguments
  1715	        x: Input tensor or variable.
  1716	        gamma: Tensor by which to scale the input.
  1717	        beta: Tensor with which to center the input.
  1718	        reduction_axes: iterable of integers,
  1719	            axes over which to normalize.
  1720	        epsilon: Fuzz factor.
  1721	
  1722	    # Returns
  1723	        A tuple length of 3, `(normalized_tensor, mean, variance)`.
  1724	    """
  1725	    mean, var = tf.nn.moments(x, reduction_axes,
  1726	                              shift=None, name=None, keep_dims=False)
  1727	    target_shape = []
  1728	    for axis in range(ndim(x)):
  1729	        if axis in reduction_axes:
  1730	            target_shape.append(1)
  1731	        else:
  1732	            target_shape.append(tf.shape(x)[axis])
  1733	    target_shape = tf.stack(target_shape)
  1734	
  1735	    broadcast_mean = tf.reshape(mean, target_shape)
  1736	    broadcast_var = tf.reshape(var, target_shape)
  1737	    if gamma is None:
  1738	        broadcast_gamma = None
  1739	    else:
  1740	        broadcast_gamma = tf.reshape(gamma, target_shape)
  1741	    if beta is None:
  1742	        broadcast_beta = None
  1743	    else:
  1744	        broadcast_beta = tf.reshape(beta, target_shape)
  1745	
  1746	    normed = tf.nn.batch_normalization(
  1747	        x,
  1748	        broadcast_mean,
  1749	        broadcast_var,
  1750	        broadcast_beta,
  1751	        broadcast_gamma,
  1752	        epsilon)
  1753	    return normed, mean, var

  1756	def _fused_normalize_batch_in_training(x, gamma, beta, reduction_axes,
  1757	                                       epsilon=1e-3):
  1758	    """Fused version of `normalize_batch_in_training`.
  1759	
  1760	    # Arguments
  1761	        x: Input tensor or variable.
  1762	        gamma: Tensor by which to scale the input.
  1763	        beta: Tensor with which to center the input.
  1764	        reduction_axes: iterable of integers,
  1765	            axes over which to normalize.
  1766	        epsilon: Fuzz factor.
  1767	
  1768	    # Returns
  1769	        A tuple length of 3, `(normalized_tensor, mean, variance)`.
  1770	    """
  1771	    if list(reduction_axes) == [0, 1, 2]:
  1772	        normalization_axis = 3
  1773	        tf_data_format = 'NHWC'
  1774	    else:
  1775	        normalization_axis = 1
  1776	        tf_data_format = 'NCHW'
  1777	
  1778	    if gamma is None:
  1779	        gamma = tf.constant(1.0,
  1780	                            dtype=x.dtype,
  1781	                            shape=[x.get_shape()[normalization_axis]])
  1782	    if beta is None:
  1783	        beta = tf.constant(0.0,
  1784	                           dtype=x.dtype,
  1785	                           shape=[x.get_shape()[normalization_axis]])
  1786	
  1787	    return tf.nn.fused_batch_norm(
  1788	        x,
  1789	        gamma,
  1790	        beta,
  1791	        epsilon=epsilon,
  1792	        data_format=tf_data_format)

  1795	def normalize_batch_in_training(x, gamma, beta,
  1796	                                reduction_axes, epsilon=1e-3):
  1797	    """Computes mean and std for batch then apply batch_normalization on batch.
  1798	
  1799	    # Arguments
  1800	        x: Input tensor or variable.
  1801	        gamma: Tensor by which to scale the input.
  1802	        beta: Tensor with which to center the input.
  1803	        reduction_axes: iterable of integers,
  1804	            axes over which to normalize.
  1805	        epsilon: Fuzz factor.
  1806	
  1807	    # Returns
  1808	        A tuple length of 3, `(normalized_tensor, mean, variance)`.
  1809	    """
  1810	    if ndim(x) == 4 and list(reduction_axes) in [[0, 1, 2], [0, 2, 3]]:
  1811	        if not _has_nchw_support() and list(reduction_axes) == [0, 2, 3]:
  1812	            return _broadcast_normalize_batch_in_training(x, gamma, beta,
  1813	                                                          reduction_axes,
  1814	                                                          epsilon=epsilon)
  1815	        return _fused_normalize_batch_in_training(
  1816	            x, gamma, beta, reduction_axes,
  1817	            epsilon=epsilon)
  1818	    else:
  1819	        if sorted(reduction_axes) == list(range(ndim(x)))[:-1]:

  1821	                                                        reduction_axes,
  1822	                                                        epsilon=epsilon)
  1823	        else:
  1824	            return _broadcast_normalize_batch_in_training(x, gamma, beta,

  1968	def repeat_elements(x, rep, axis):
  1969	    """Repeats the elements of a tensor along an axis, like `np.repeat`.
  1970	
  1971	    If `x` has shape `(s1, s2, s3)` and `axis` is `1`, the output
  1972	    will have shape `(s1, s2 * rep, s3)`.
  1973	
  1974	    # Arguments
  1975	        x: Tensor or variable.
  1976	        rep: Python integer, number of times to repeat.
  1977	        axis: Axis along which to repeat.
  1978	
  1979	    # Returns
  1980	        A tensor.
  1981	    """
  1982	    x_shape = x.get_shape().as_list()
  1983	    # For static axis
  1984	    if x_shape[axis] is not None:
  1985	        # slices along the repeat axis
  1986	        splits = tf.split(value=x, num_or_size_splits=x_shape[axis], axis=axis)
  1987	        # repeat each slice the given number of reps
  1988	        x_rep = [s for s in splits for _ in range(rep)]
  1989	        return concatenate(x_rep, axis)
  1990	
  1991	    # Here we use tf.tile to mimic behavior of np.repeat so that
  1992	    # we can handle dynamic shapes (that include None).
  1993	    # To do that, we need an auxiliary axis to repeat elements along
  1994	    # it and then merge them along the desired axis.
  1995	
  1996	    # Repeating
  1997	    auxiliary_axis = axis + 1
  1998	    x_shape = tf.shape(x)
  1999	    x_rep = tf.expand_dims(x, axis=auxiliary_axis)
  2000	    reps = np.ones(len(x.get_shape()) + 1)
  2001	    reps[auxiliary_axis] = rep
  2002	    x_rep = tf.tile(x_rep, reps)
  2003	
  2004	    # Merging
  2005	    reps = np.delete(reps, auxiliary_axis)
  2006	    reps[axis] = rep
  2007	    reps = tf.constant(reps, dtype='int32')
  2008	    x_shape = x_shape * reps
  2009	    x_rep = tf.reshape(x_rep, x_shape)
  2010	
  2011	    # Fix shape representation
  2012	    x_shape = x.get_shape().as_list()
  2013	    x_rep.set_shape(x_shape)
  2014	    x_rep._keras_shape = tuple(x_shape)
  2015	    return x_rep

  2037	def arange(start, stop=None, step=1, dtype='int32'):
  2038	    """Creates a 1D tensor containing a sequence of integers.
  2039	
  2040	    The function arguments use the same convention as
  2041	    Theano's arange: if only one argument is provided,
  2042	    it is in fact the "stop" argument.

  2057	    # Match the behavior of numpy and Theano by returning an empty seqence.

  2063	            # Handle case where start is a tensor

  2066	                            false_fn=lambda: start)

  2402	class Function(object):
  2403	    """Runs a computation graph.
  2404	
  2405	    It's possible to pass arguments to `tf.Session.run()` via `session_kwargs`.
  2406	    In particular additional operations via `fetches` argument and additional
  2407	    tensor substitutions via `feed_dict` arguments. Note that given
  2408	    substitutions are merged with substitutions from `inputs`. Even though
  2409	    `feed_dict` is passed once in the constructor (called in `model.compile()`)
  2410	    we can modify the values in the dictionary. Through this feed_dict we can
  2411	    provide additional substitutions besides Keras inputs.
  2412	
  2413	    # Arguments
  2414	        inputs: Feed placeholders to the computation graph.
  2415	        outputs: Output tensors to fetch.
  2416	        updates: Additional update ops to be run at function call.
  2417	        name: a name to help users identify what this function does.
  2418	        session_kwargs: arguments to `tf.Session.run()`: `fetches`, `feed_dict`,
  2419	        `options`, `run_metadata`
  2420	    """
  2421	
  2422	    def __init__(self, inputs, outputs, updates=None, name=None, **session_kwargs):
  2423	        updates = updates or []
  2424	        if not isinstance(inputs, (list, tuple)):
  2425	            raise TypeError('`inputs` to a TensorFlow backend function '
  2426	                            'should be a list or tuple.')
  2427	        if not isinstance(outputs, (list, tuple)):
  2428	            raise TypeError('`outputs` of a TensorFlow backend function '
  2429	                            'should be a list or tuple.')
  2430	        if not isinstance(updates, (list, tuple)):
  2431	            raise TypeError('`updates` in a TensorFlow backend function '
  2432	                            'should be a list or tuple.')
  2433	        self.inputs = list(inputs)
  2434	        self.outputs = list(outputs)
  2435	        with tf.control_dependencies(self.outputs):
  2436	            updates_ops = []
  2437	            for update in updates:
  2438	                if isinstance(update, tuple):
  2439	                    p, new_p = update
  2440	                    updates_ops.append(tf.assign(p, new_p))
  2441	                else:
  2442	                    # assumed already an op
  2443	                    updates_ops.append(update)
  2444	            self.updates_op = tf.group(*updates_ops)
  2445	        self.name = name
  2446	        # additional tensor substitutions
  2447	        self.feed_dict = session_kwargs.pop('feed_dict', {})
  2448	        # additional operations
  2449	        self.fetches = session_kwargs.pop('fetches', [])
  2450	        if not isinstance(self.fetches, list):
  2451	            self.fetches = [self.fetches]
  2452	        self.session_kwargs = session_kwargs
  2453	
  2454	    def __call__(self, inputs):
  2455	        if not isinstance(inputs, (list, tuple)):
  2456	            raise TypeError('`inputs` should be a list or tuple.')
  2457	        feed_dict = self.feed_dict.copy()
  2458	        for tensor, value in zip(self.inputs, inputs):
  2459	            if is_sparse(tensor):
  2460	                sparse_coo = value.tocoo()
  2461	                indices = np.concatenate((np.expand_dims(sparse_coo.row, 1),
  2462	                                          np.expand_dims(sparse_coo.col, 1)), 1)
  2463	                value = (indices, sparse_coo.data, sparse_coo.shape)
  2464	            feed_dict[tensor] = value
  2465	        fetches = self.outputs + [self.updates_op] + self.fetches
  2466	        session = get_session()
  2467	        updated = session.run(fetches=fetches, feed_dict=feed_dict,
  2468	                              **self.session_kwargs)
  2469	        return updated[:len(self.outputs)]
  2470

  2472	def function(inputs, outputs, updates=None, **kwargs):
  2473	    """Instantiates a Keras function.
  2474	
  2475	    # Arguments
  2476	        inputs: List of placeholder tensors.
  2477	        outputs: List of output tensors.
  2478	        updates: List of update ops.
  2479	        **kwargs: Passed to `tf.Session.run`.

  2490	                msg = 'Invalid argument "%s" passed to K.function with TensorFlow backend' % key

  2527	def rnn(step_function, inputs, initial_states,
  2528	        go_backwards=False, mask=None, constants=None,
  2529	        unroll=False, input_length=None):
  2530	    """Iterates over the time dimension of a tensor.
  2531	
  2532	    # Arguments
  2533	        step_function: RNN step function.
  2534	            Parameters:
  2535	                inputs: tensor with shape `(samples, ...)` (no time dimension),
  2536	                    representing input for the batch of samples at a certain
  2537	                    time step.
  2538	                states: list of tensors.
  2539	            Returns:
  2540	                outputs: tensor with shape `(samples, output_dim)`
  2541	                    (no time dimension).
  2542	                new_states: list of tensors, same length and shapes
  2543	                    as 'states'. The first state in the list must be the
  2544	                    output tensor at the previous timestep.
  2545	        inputs: tensor of temporal data of shape `(samples, time, ...)`
  2546	            (at least 3D).
  2547	        initial_states: tensor with shape (samples, output_dim)
  2548	            (no time dimension),
  2549	            containing the initial values for the states used in
  2550	            the step function.
  2551	        go_backwards: boolean. If True, do the iteration over the time
  2552	            dimension in reverse order and return the reversed sequence.
  2553	        mask: binary tensor with shape `(samples, time, 1)`,
  2554	            with a zero for every element that is masked.
  2555	        constants: a list of constant values passed at each step.
  2556	        unroll: whether to unroll the RNN or to use a symbolic loop (`while_loop` or `scan` depending on backend).
  2557	        input_length: not relevant in the TensorFlow implementation.
  2558	            Must be specified if using unrolling with Theano.
  2559	
  2560	    # Returns
  2561	        A tuple, `(last_output, outputs, new_states)`.
  2562	
  2563	            last_output: the latest output of the rnn, of shape `(samples, ...)`
  2564	            outputs: tensor with shape `(samples, time, ...)` where each
  2565	                entry `outputs[s, t]` is the output of the step function
  2566	                at time `t` for sample `s`.
  2567	            new_states: list of tensors, latest states returned by
  2568	                the step function, of shape `(samples, ...)`.
  2569	
  2570	    # Raises
  2571	        ValueError: if input dimension is less than 3.
  2572	        ValueError: if `unroll` is `True` but input timestep is not a fixed number.
  2573	        ValueError: if `mask` is provided (not `None`) but states is not provided
  2574	            (`len(states)` == 0).
  2575	    """
  2576	    ndim = len(inputs.get_shape())
  2577	    if ndim < 3:
  2578	        raise ValueError('Input should be at least 3D.')
  2579	
  2580	    # Transpose to time-major, i.e.
  2581	    # from (batch, time, ...) to (time, batch, ...)
  2582	    axes = [1, 0] + list(range(2, ndim))
  2583	    inputs = tf.transpose(inputs, (axes))
  2584	
  2585	    if mask is not None:
  2586	        if mask.dtype != tf.bool:
  2587	            mask = tf.cast(mask, tf.bool)
  2588	        if len(mask.get_shape()) == ndim - 1:
  2589	            mask = expand_dims(mask)
  2590	        mask = tf.transpose(mask, axes)
  2591	
  2592	    if constants is None:
  2593	        constants = []
  2594	
  2595	    global uses_learning_phase
  2596	    uses_learning_phase = False
  2597	
  2598	    if unroll:
  2599	        if not inputs.get_shape()[0]:
  2600	            raise ValueError('Unrolling requires a '
  2601	                             'fixed number of timesteps.')
  2602	        states = initial_states
  2603	        successive_states = []
  2604	        successive_outputs = []
  2605	
  2606	        input_list = tf.unstack(inputs)
  2607	        if go_backwards:
  2608	            input_list.reverse()
  2609	
  2610	        if mask is not None:
  2611	            mask_list = tf.unstack(mask)
  2612	            if go_backwards:
  2613	                mask_list.reverse()
  2614	
  2615	            for inp, mask_t in zip(input_list, mask_list):
  2616	                output, new_states = step_function(inp, states + constants)
  2617	                if getattr(output, '_uses_learning_phase', False):
  2618	                    uses_learning_phase = True
  2619	
  2620	                # tf.where needs its condition tensor
  2621	                # to be the same shape as its two
  2622	                # result tensors, but in our case
  2623	                # the condition (mask) tensor is
  2624	                # (nsamples, 1), and A and B are (nsamples, ndimensions).
  2625	                # So we need to
  2626	                # broadcast the mask to match the shape of A and B.
  2627	                # That's what the tile call does,
  2628	                # it just repeats the mask along its second dimension
  2629	                # n times.
  2630	                tiled_mask_t = tf.tile(mask_t,
  2631	                                       tf.stack([1, tf.shape(output)[1]]))
  2632	
  2633	                if not successive_outputs:
  2634	                    prev_output = zeros_like(output)
  2635	                else:
  2636	                    prev_output = successive_outputs[-1]
  2637	
  2638	                output = tf.where(tiled_mask_t, output, prev_output)
  2639	
  2640	                return_states = []
  2641	                for state, new_state in zip(states, new_states):
  2642	                    # (see earlier comment for tile explanation)
  2643	                    tiled_mask_t = tf.tile(mask_t,
  2644	                                           tf.stack([1, tf.shape(new_state)[1]]))
  2645	                    return_states.append(tf.where(tiled_mask_t,
  2646	                                                  new_state,
  2647	                                                  state))
  2648	                states = return_states
  2649	                successive_outputs.append(output)
  2650	                successive_states.append(states)
  2651	            last_output = successive_outputs[-1]
  2652	            new_states = successive_states[-1]
  2653	            outputs = tf.stack(successive_outputs)
  2654	        else:
  2655	            for inp in input_list:
  2656	                output, states = step_function(inp, states + constants)
  2657	                if getattr(output, '_uses_learning_phase', False):
  2658	                    uses_learning_phase = True
  2659	                successive_outputs.append(output)
  2660	                successive_states.append(states)
  2661	            last_output = successive_outputs[-1]
  2662	            new_states = successive_states[-1]
  2663	            outputs = tf.stack(successive_outputs)
  2664	
  2665	    else:
  2666	        if go_backwards:
  2667	            inputs = reverse(inputs, 0)
  2668	
  2669	        states = tuple(initial_states)
  2670	
  2671	        time_steps = tf.shape(inputs)[0]
  2672	        outputs, _ = step_function(inputs[0], initial_states + constants)
  2673	        output_ta = tensor_array_ops.TensorArray(
  2674	            dtype=outputs.dtype,
  2675	            size=time_steps,
  2676	            tensor_array_name='output_ta')
  2677	        input_ta = tensor_array_ops.TensorArray(
  2678	            dtype=inputs.dtype,
  2679	            size=time_steps,
  2680	            tensor_array_name='input_ta')
  2681	        input_ta = input_ta.unstack(inputs)
  2682	        time = tf.constant(0, dtype='int32', name='time')
  2683	
  2684	        if mask is not None:
  2685	            if not states:
  2686	                raise ValueError('No initial states provided! '
  2687	                                 'When using masking in an RNN, you should '
  2688	                                 'provide initial states '
  2689	                                 '(and your step function should return '
  2690	                                 'as its first state at time `t` '
  2691	                                 'the output at time `t-1`).')
  2692	            if go_backwards:
  2693	                mask = reverse(mask, 0)
  2694	
  2695	            mask_ta = tensor_array_ops.TensorArray(
  2696	                dtype=tf.bool,
  2697	                size=time_steps,
  2698	                tensor_array_name='mask_ta')
  2699	            mask_ta = mask_ta.unstack(mask)
  2700	
  2701	            def _step(time, output_ta_t, *states):
  2702	                """RNN step function.
  2703	
  2704	                # Arguments
  2705	                    time: Current timestep value.
  2706	                    output_ta_t: TensorArray.
  2707	                    *states: List of states.
  2708	
  2709	                # Returns
  2710	                    Tuple: `(time + 1,output_ta_t) + tuple(new_states)`
  2711	                """
  2712	                current_input = input_ta.read(time)
  2713	                mask_t = mask_ta.read(time)
  2714	                output, new_states = step_function(current_input,
  2715	                                                   tuple(states) +
  2716	                                                   tuple(constants))
  2717	                if getattr(output, '_uses_learning_phase', False):
  2718	                    global uses_learning_phase
  2719	                    uses_learning_phase = True
  2720	                for state, new_state in zip(states, new_states):
  2721	                    new_state.set_shape(state.get_shape())
  2722	                tiled_mask_t = tf.tile(mask_t,
  2723	                                       tf.stack([1, tf.shape(output)[1]]))
  2724	                output = tf.where(tiled_mask_t, output, states[0])
  2725	                new_states = [tf.where(tiled_mask_t, new_states[i], states[i]) for i in range(len(states))]
  2726	                output_ta_t = output_ta_t.write(time, output)
  2727	                return (time + 1, output_ta_t) + tuple(new_states)
  2728	        else:
  2729	            def _step(time, output_ta_t, *states):
  2730	                """RNN step function.
  2731	
  2732	                # Arguments
  2733	                    time: Current timestep value.
  2734	                    output_ta_t: TensorArray.
  2735	                    *states: List of states.
  2736	
  2737	                # Returns
  2738	                    Tuple: `(time + 1,output_ta_t) + tuple(new_states)`
  2739	                """
  2740	                current_input = input_ta.read(time)
  2741	                output, new_states = step_function(current_input,
  2742	                                                   tuple(states) +
  2743	                                                   tuple(constants))
  2744	                if getattr(output, '_uses_learning_phase', False):
  2745	                    global uses_learning_phase
  2746	                    uses_learning_phase = True
  2747	                for state, new_state in zip(states, new_states):
  2748	                    new_state.set_shape(state.get_shape())
  2749	                output_ta_t = output_ta_t.write(time, output)
  2750	                return (time + 1, output_ta_t) + tuple(new_states)
  2751	
  2752	        final_outputs = control_flow_ops.while_loop(
  2753	            cond=lambda time, *_: time < time_steps,
  2754	            body=_step,
  2755	            loop_vars=(time, output_ta) + states,
  2756	            parallel_iterations=32,
  2757	            swap_memory=True)
  2758	        last_time = final_outputs[0]
  2759	        output_ta = final_outputs[1]
  2760	        new_states = final_outputs[2:]
  2761	
  2762	        outputs = output_ta.stack()
  2763	        last_output = output_ta.read(last_time - 1)
  2764	
  2765	    axes = [1, 0] + list(range(2, len(outputs.get_shape())))
  2766	    outputs = tf.transpose(outputs, axes)
  2767	    last_output._uses_learning_phase = uses_learning_phase
  2768	    return last_output, outputs, new_states

  2771	def switch(condition, then_expression, else_expression):
  2772	    """Switches between two operations depending on a scalar value.
  2773	
  2774	    Note that both `then_expression` and `else_expression`
  2775	    should be symbolic tensors of the *same shape*.
  2776	
  2777	    # Arguments
  2778	        condition: tensor (`int` or `bool`).
  2779	        then_expression: either a tensor, or a callable that returns a tensor.
  2780	        else_expression: either a tensor, or a callable that returns a tensor.
  2781	
  2782	    # Returns
  2783	        The selected tensor.
  2784	
  2785	    # Raises
  2786	        ValueError: If rank of `condition` is greater than rank of expressions.
  2787	    """
  2788	    if condition.dtype != tf.bool:
  2789	        condition = tf.cast(condition, 'bool')
  2790	    cond_ndim = ndim(condition)
  2791	    if not cond_ndim:
  2792	        if not callable(then_expression):
  2793	            def then_expression_fn():
  2794	                return then_expression
  2795	        else:
  2796	            then_expression_fn = then_expression
  2797	        if not callable(else_expression):
  2798	            def else_expression_fn():
  2799	                return else_expression
  2800	        else:
  2801	            else_expression_fn = else_expression
  2802	        x = tf.cond(condition,
  2803	                    then_expression_fn,
  2804	                    else_expression_fn)
  2805	    else:
  2806	        # tf.where needs its condition tensor
  2807	        # to be the same shape as its two
  2808	        # result tensors
  2809	        if callable(then_expression):
  2810	            then_expression = then_expression()
  2811	        if callable(else_expression):
  2812	            else_expression = else_expression()
  2813	        expr_ndim = ndim(then_expression)
  2814	        if cond_ndim > expr_ndim:
  2815	            raise ValueError('Rank of `condition` should be less than or'
  2816	                             ' equal to rank of `then_expression` and '
  2817	                             '`else_expression`. ndim(condition)=' +
  2818	                             str(cond_ndim) + ', ndim(then_expression)'
  2819	                             '=' + str(expr_ndim))
  2820	        if cond_ndim > 1:
  2821	            ndim_diff = expr_ndim - cond_ndim
  2822	            cond_shape = tf.concat([tf.shape(condition), [1] * ndim_diff], axis=0)
  2823	            condition = tf.reshape(condition, cond_shape)
  2824	            expr_shape = tf.shape(then_expression)
  2825	            shape_diff = expr_shape - cond_shape
  2826	            tile_shape = tf.where(shape_diff > 0, expr_shape, tf.ones_like(expr_shape))
  2827	            condition = tf.tile(condition, tile_shape)
  2828	        x = tf.where(condition, then_expression, else_expression)
  2829	    return x

  2832	def in_train_phase(x, alt, training=None):
  2833	    """Selects `x` in train phase, and `alt` otherwise.
  2834	
  2835	    Note that `alt` should have the *same shape* as `x`.

  2898	    """Rectified linear unit.
  2899

  2910	    if alpha != 0.:

  2977	    """Categorical crossentropy between an output tensor and a target tensor.

  2979	    # Arguments

  3001	                               axis=len(output.get_shape()) - 1)

  3713	
  3714	    # Arguments
  3715	        x: Tensor or variable.

  3720	        Output tensor.

  3729	    if data_format is None:
  3730	        data_format = image_data_format()

  3735	        raise ValueError('Unexpected bias dimensions %d, expect to be 1 or %d dimensions'
  3736	                         % (len(bias_shape), ndim(x)))
  3737	    if ndim(x) == 5:
  3738	        if data_format == 'channels_first':
  3739	            if len(bias_shape) == 1:
  3740	                x += reshape(bias, (1, bias_shape[0], 1, 1, 1))

  3743	        elif data_format == 'channels_last':
  3744	            if len(bias_shape) == 1:
  3745	                x += reshape(bias, (1, 1, 1, bias_shape[0]))
  3746	            else:
  3747	                x += reshape(bias, (1,) + bias_shape)

  3753	                                       data_format='NCHW')

  3756	            else:
  3757	                x += reshape(bias, (1, bias_shape[2]) + bias_shape[:2])
  3758	        elif data_format == 'channels_last':

  3761	                                   data_format='NHWC')

  3873	# CTC
  3874	# TensorFlow has a native implementation, but it uses sparse tensors
  3875	# and therefore requires a wrapper for Keras. The functions below convert
  3876	# dense to sparse tensors and also wraps up the beam search code that is
  3877	# in TensorFlow's CTC implementation

  3880	def ctc_label_dense_to_sparse(labels, label_lengths):
  3881	    """Converts CTC labels from dense to sparse.
  3882	
  3883	    # Arguments
  3884	        labels: dense CTC labels.
  3885	        label_lengths: length of the labels.
  3886	
  3887	    # Returns
  3888	        A sparse tensor representation of the labels.
  3889	    """
  3890	    label_shape = tf.shape(labels)
  3891	    num_batches_tns = tf.stack([label_shape[0]])
  3892	    max_num_labels_tns = tf.stack([label_shape[1]])
  3893	
  3894	    def range_less_than(_, current_input):
  3895	        return tf.expand_dims(tf.range(label_shape[1]), 0) < tf.fill(
  3896	            max_num_labels_tns, current_input)
  3897	
  3898	    init = tf.cast(tf.fill([1, label_shape[1]], 0), tf.bool)
  3899	    dense_mask = functional_ops.scan(range_less_than, label_lengths,
  3900	                                     initializer=init, parallel_iterations=1)
  3901	    dense_mask = dense_mask[:, 0, :]
  3902	
  3903	    label_array = tf.reshape(tf.tile(tf.range(label_shape[1]), num_batches_tns),
  3904	                             label_shape)
  3905	    label_ind = tf.boolean_mask(label_array, dense_mask)
  3906	
  3907	    batch_array = tf.transpose(tf.reshape(tf.tile(tf.range(label_shape[0]),
  3908	                                                  max_num_labels_tns), reverse(label_shape, 0)))
  3909	    batch_ind = tf.boolean_mask(batch_array, dense_mask)
  3910	    indices = tf.transpose(tf.reshape(concatenate([batch_ind, label_ind], axis=0), [2, -1]))
  3911	
  3912	    vals_sparse = tf.gather_nd(labels, indices)
  3913	
  3914	    return tf.SparseTensor(tf.to_int64(indices), vals_sparse, tf.to_int64(label_shape))

  3917	def ctc_batch_cost(y_true, y_pred, input_length, label_length):
  3918	    """Runs CTC loss algorithm on each batch element.
  3919	
  3920	    # Arguments
  3921	        y_true: tensor `(samples, max_string_length)`
  3922	            containing the truth labels.
  3923	        y_pred: tensor `(samples, time_steps, num_categories)`
  3924	            containing the prediction, or output of the softmax.
  3925	        input_length: tensor `(samples, 1)` containing the sequence length for
  3926	            each batch item in `y_pred`.
  3927	        label_length: tensor `(samples, 1)` containing the sequence length for
  3928	            each batch item in `y_true`.
  3929	
  3930	    # Returns
  3931	        Tensor with shape (samples,1) containing the
  3932	            CTC loss of each element.
  3933	    """
  3934	    label_length = tf.to_int32(tf.squeeze(label_length))
  3935	    input_length = tf.to_int32(tf.squeeze(input_length))
  3936	    sparse_labels = tf.to_int32(ctc_label_dense_to_sparse(y_true, label_length))
  3937	
  3938	    y_pred = tf.log(tf.transpose(y_pred, perm=[1, 0, 2]) + epsilon())
  3939	
  3940	    return tf.expand_dims(ctc.ctc_loss(inputs=y_pred,
  3941	                                       labels=sparse_labels,
  3942	                                       sequence_length=input_length), 1)

  3945	def ctc_decode(y_pred, input_length, greedy=True, beam_width=100,
  3946	               top_paths=1):
  3947	    """Decodes the output of a softmax.
  3948	
  3949	    Can use either greedy search (also known as best path)
  3950	    or a constrained dictionary search.
  3951	
  3952	    # Arguments
  3953	        y_pred: tensor `(samples, time_steps, num_categories)`
  3954	            containing the prediction, or output of the softmax.
  3955	        input_length: tensor `(samples, )` containing the sequence length for
  3956	            each batch item in `y_pred`.
  3957	        greedy: perform much faster best-path search if `true`.
  3958	            This does not use a dictionary.
  3959	        beam_width: if `greedy` is `false`: a beam search decoder will be used
  3960	            with a beam of this width.
  3961	        top_paths: if `greedy` is `false`,
  3962	            how many of the most probable paths will be returned.
  3963	
  3964	    # Returns
  3965	        Tuple:
  3966	            List: if `greedy` is `true`, returns a list of one element that
  3967	                contains the decoded sequence.
  3968	                If `false`, returns the `top_paths` most probable
  3969	                decoded sequences.
  3970	                Important: blank labels are returned as `-1`.
  3971	            Tensor `(top_paths, )` that contains
  3972	                the log probability of each decoded sequence.
  3973	    """
  3974	    y_pred = tf.log(tf.transpose(y_pred, perm=[1, 0, 2]) + epsilon())
  3975	    input_length = tf.to_int32(input_length)
  3976	
  3977	    if greedy:
  3978	        (decoded, log_prob) = ctc.ctc_greedy_decoder(
  3979	            inputs=y_pred,
  3980	            sequence_length=input_length)
  3981	    else:
  3982	        (decoded, log_prob) = ctc.ctc_beam_search_decoder(
  3983	            inputs=y_pred,
  3984	            sequence_length=input_length, beam_width=beam_width,
  3985	            top_paths=top_paths)
  3986	
  3987	    decoded_dense = [tf.sparse_to_dense(st.indices, st.dense_shape, st.values, default_value=-1)
  3988	                     for st in decoded]
  3989	    return (decoded_dense, log_prob)

  4041	def local_conv1d(inputs, kernel, kernel_size, strides, data_format=None):
  4042	    """Apply 1D conv with un-shared weights.
  4043	
  4044	    # Arguments
  4045	        inputs: 3D tensor with shape: (batch_size, steps, input_dim)
  4046	        kernel: the unshared weight for convolution,
  4047	                with shape (output_length, feature_dim, filters)
  4048	        kernel_size: a tuple of a single integer,
  4049	                     specifying the length of the 1D convolution window
  4050	        strides: a tuple of a single integer,
  4051	                 specifying the stride length of the convolution
  4052	        data_format: the data format, channels_first or channels_last
  4053	
  4054	    # Returns
  4055	        the tensor after 1d conv with un-shared weights, with shape (batch_size, output_length, filters)
  4056	
  4057	    # Raises
  4058	        ValueError: if `data_format` is neither `channels_last` or `channels_first`.
  4059	    """
  4060	    if data_format is None:
  4061	        data_format = image_data_format()
  4062	    if data_format not in {'channels_first', 'channels_last'}:
  4063	        raise ValueError('Unknown data_format ' + str(data_format))
  4064	
  4065	    stride = strides[0]
  4066	    kernel_shape = int_shape(kernel)
  4067	    output_length, feature_dim, filters = kernel_shape
  4068	
  4069	    xs = []
  4070	    for i in range(output_length):
  4071	        slice_length = slice(i * stride,
  4072	                             i * stride + kernel_size[0])
  4073	        xs.append(reshape(inputs[:, slice_length, :],
  4074	                          (1, -1, feature_dim)))
  4075	    x_aggregate = concatenate(xs, axis=0)
  4076	    # Shape: `(output_length, batch_size, filters)`.
  4077	    output = batch_dot(x_aggregate, kernel)
  4078	    return permute_dimensions(output, (1, 0, 2))

  4081	def local_conv2d(inputs, kernel, kernel_size, strides, output_shape, data_format=None):
  4082	    """Apply 2D conv with un-shared weights.
  4083	
  4084	    # Arguments
  4085	        inputs: 4D tensor with shape:
  4086	                (batch_size, filters, new_rows, new_cols)
  4087	                if data_format='channels_first'
  4088	                or 4D tensor with shape:
  4089	                (batch_size, new_rows, new_cols, filters)
  4090	                if data_format='channels_last'.
  4091	        kernel: the unshared weight for convolution,
  4092	                with shape (output_items, feature_dim, filters)
  4093	        kernel_size: a tuple of 2 integers, specifying the
  4094	                     width and height of the 2D convolution window.
  4095	        strides: a tuple of 2 integers, specifying the strides
  4096	                 of the convolution along the width and height.
  4097	        output_shape: a tuple with (output_row, output_col)
  4098	        data_format: the data format, channels_first or channels_last
  4099	
  4100	    # Returns
  4101	        A 4d tensor with shape:
  4102	        (batch_size, filters, new_rows, new_cols)
  4103	        if data_format='channels_first'
  4104	        or 4D tensor with shape:
  4105	        (batch_size, new_rows, new_cols, filters)
  4106	        if data_format='channels_last'.
  4107	
  4108	    # Raises
  4109	        ValueError: if `data_format` is neither
  4110	                    `channels_last` or `channels_first`.
  4111	    """
  4112	    if data_format is None:
  4113	        data_format = image_data_format()
  4114	    if data_format not in {'channels_first', 'channels_last'}:
  4115	        raise ValueError('Unknown data_format ' + str(data_format))
  4116	
  4117	    stride_row, stride_col = strides
  4118	    output_row, output_col = output_shape
  4119	    kernel_shape = int_shape(kernel)
  4120	    _, feature_dim, filters = kernel_shape
  4121	
  4122	    xs = []
  4123	    for i in range(output_row):
  4124	        for j in range(output_col):
  4125	            slice_row = slice(i * stride_row,
  4126	                              i * stride_row + kernel_size[0])
  4127	            slice_col = slice(j * stride_col,
  4128	                              j * stride_col + kernel_size[1])
  4129	            if data_format == 'channels_first':
  4130	                xs.append(reshape(inputs[:, :, slice_row, slice_col],
  4131	                                  (1, -1, feature_dim)))
  4132	            else:
  4133	                xs.append(reshape(inputs[:, slice_row, slice_col, :],
  4134	                                  (1, -1, feature_dim)))
  4135	
  4136	    x_aggregate = concatenate(xs, axis=0)
  4137	    output = batch_dot(x_aggregate, kernel)
  4138	    output = reshape(output,
  4139	                     (output_row, output_col, -1, filters))
  4140	
  4141	    if data_format == 'channels_first':
  4142	        output = permute_dimensions(output, (2, 3, 0, 1))
  4143	    else:
  4144	        output = permute_dimensions(output, (2, 0, 1, 3))
  4145	    return output
