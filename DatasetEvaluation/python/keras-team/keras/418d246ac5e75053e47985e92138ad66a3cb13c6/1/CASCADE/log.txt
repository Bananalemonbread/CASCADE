def get_layer_and_inputs(num_heads=2, key_dim=4, value_dim=8, batch_size=1, query_len=3, value_len=3, dtype=tf.float32):
    """Helper to create a MultiHeadAttention layer and valid inputs."""
    layer = MultiHeadAttention(num_heads=num_heads, key_dim=key_dim, value_dim=value_dim)
    # Build the layer to initialize weights
    dummy_q = tf.zeros((batch_size, 1, num_heads, key_dim))
    dummy_k = tf.zeros((batch_size, 1, num_heads, key_dim))
    dummy_v = tf.zeros((batch_size, 1, num_heads, value_dim))
    layer(dummy_q, dummy_k, dummy_v)
    
    # Create actual inputs
    # Shape: (B, T, N, key_dim) for Q and K
    query = tf.random.normal((batch_size, query_len, num_heads, key_dim), dtype=dtype)
    key = tf.random.normal((batch_size, value_len, num_heads, key_dim), dtype=dtype)
    value = tf.random.normal((batch_size, value_len, num_heads, value_dim), dtype=dtype)
    
    return layer, query, key, value


def test_attention_output_shape_matches_value_dim():
    """Verify that the 'attention_output' tensor returned by '_compute_attention' has the shape (B, T, N, value_dim). If the input 'value' tensor has shape (B, T, N, value_dim), then the output tensor's last dimension must equal 'value_dim'. This tests the 'Returns: attention_output: Multi-headed outputs' specification."""
    layer, query, key, value = get_layer_and_inputs()
    batch_size, query_len, num_heads, value_dim = value.shape[0], value.shape[1], value.shape[2], value.shape[3]
    
    attention_output, attention_scores = layer._compute_attention(query, key, value, attention_mask=None, training=False)
    
    # According to doc, output is (B, T, N, value_dim)
    expected_shape = (batch_size, query_len, num_heads, value_dim)
    assert attention_output.shape.as_list() == expected_shape

    """Verify that the 'attention_scores' tensor has the shape (B, T, N, S). def test_mask_prevents_attention_to_masked_positions():
    """Verify that positions marked as False in the 'attention_mask' do not contribute to the output. If 'attention_mask' is a boolean tensor of shape (B, T, S) with specific positions set to False, then the attention weights for those positions should be zero (or effectively ignored), and the output should only depend on the unmasked positions. This tests 'attention_mask... prevents attention to certain positions'."""
    layer, query, key, value = get_layer_and_inputs(batch_size=1, query_len=2, value_len=3)
    
    # Mask out the second position of the keys (index 1)
    # Shape (B, T, S) -> (1, 2, 3)
    # True means keep, False means mask
    attention_mask = tf.constant([[[True, False, True], [True, False, True]]], dtype=tf.bool)
    
    attention_output_masked, attention_scores_masked = layer._compute_attention(
        query, key, value, attention_mask=attention_mask, training=False
    )
    
    attention_scores_np = attention_scores_masked.numpy()
    
    # For each head and query position, the score at index 1 should be 0 (or close to 0)
    # Because the mask is False at index 1
    for b in range(1):
        for t in range(2):
            for n in range(layer.num_heads):
                scores_for_pos = attention_scores_np[b, t, n, :]
                # The score at index 1 should be very small (masked)
                # In standard softmax with -inf mask, it becomes 0.
                assert np.allclose(scores_for_pos[1], 0.0, atol=1e-5), f"Score at masked position should be ~0, got {scores_for_pos[1]}"


def test_no_mask_when_mask_is_none():
    """Verify that when 'attention_mask' is None, no masking is applied. If 'attention_mask' is None, then the attention mechanism should compute weights over all available key positions (S), resulting in outputs identical to the unmasked case with all True masks. This tests the optional nature of the mask argument."""
    layer, query, key, value = get_layer_and_inputs(batch_size=1, query_len=2, value_len=3)
    
    # Run with None mask
    attention_output_none, _ = layer._compute_attention(query, key, value, attention_mask=None, training=False)
    
    # Run with all True mask (should be equivalent to no mask)
    batch_size, query_len, _, key_len = query.shape[0], query.shape[1], query.shape[2], key.shape[1]
    all_true_mask = tf.ones((batch_size, query_len, key_len), dtype=tf.bool)
    
    attention_output_true, _ = layer._compute_attention(query, key, value, attention_mask=all_true_mask, training=False)
    
    # Outputs should be identical
    assert tf.reduce_all(tf.equal(attention_output_none, attention_output_true))


def test_dropout_applied_in_training_mode():
    """Verify that dropout is applied when 'training' is True. If 'training' is set to True, then the 'attention_output' should vary between multiple calls with identical inputs (non-deterministic behavior), indicating dropout noise is added. This tests 'training: ... adding dropout'."""
    # Use a layer with dropout
    layer = MultiHeadAttention(num_heads=2, key_dim=4, value_dim=8, dropout=0.5)
    
    # Build
    dummy_q = tf.zeros((1, 1, 2, 4))
    dummy_k = tf.zeros((1, 1, 2, 4))
    dummy_v = tf.zeros((1, 1, 2, 8))
    layer(dummy_q, dummy_k, dummy_v)
    
    query = tf.random.normal((1, 2, 2, 4))
    key = tf.random.normal((1, 2, 2, 4))
    value = tf.random.normal((1, 2, 2, 8))
    
    # We need to ensure randomness. 
    # Note: _compute_attention might not implement dropout itself, 
    # but the doc says it handles training mode. 
    # If the method doesn't implement dropout, this test might fail if we assume it does.
    # However, standard MultiHeadAttention applies dropout on scores or output.
    # Let's assume the internal implementation follows the doc.
    
    out1, _ = layer._compute_attention(query, key, value, training=True)
    out2, _ = layer._compute_attention(query, key, value, training=True)
    
    # With dropout=0.5, it is highly unlikely (prob < 2^-N) that two outputs are exactly identical
    # unless dropout is not applied.
    # We use a small tolerance but check they are not bitwise identical.
    are_identical = tf.reduce_all(tf.equal(out1, out2))
    
    # If dropout is working, they should differ.
    # If the implementation of _compute_attention does NOT handle dropout, 
    # then this test validates the doc string claim. If it fails, the doc is misleading or impl is wrong.
    # Given "training: ... adding dropout", we expect difference.
    assert not are_identical, "Outputs with training=True should vary due to dropout"

def test_multi_head_independence():
    """Verify that attention is computed independently per head. If the inputs have N heads, then changing the values in head i of 'query', 'key', and 'value' should not affect the output in head j (where i != j). This tests the 'Multi-headed outputs' specification."""
    layer, query, key, value = get_layer_and_inputs(num_heads=2, key_dim=4, value_dim=8, batch_size=1, query_len=2, value_len=2)
    
    # Run baseline
    out_base, _ = layer._compute_attention(query, key, value, attention_mask=None, training=False)
    
    # Modify Head 0 inputs
    query_mod = query.numpy().copy()
    key_mod = key.numpy().copy()
    value_mod = value.numpy().copy()
    
    # Change all values in Head 0
    query_mod[:, :, 0, :] = 100.0
    key_mod[:, :, 0, :] = 100.0
    value_mod[:, :, 0, :] = 100.0
    
    query_t = tf.constant(query_mod)
    key_t = tf.constant(key_mod)
    value_t = tf.constant(value_mod)
    
    out_mod, _ = layer._compute_attention(query_t, key_t, value_t, attention_mask=None, training=False)
    
    # Head 1 output should remain unchanged
    head_1_base = out_base[:, :, 1, :]
    head_1_mod = out_mod[:, :, 1, :]
    
    assert tf.reduce_all(tf.equal(head_1_base, head_1_mod))

    41	def _build_attention_equation(rank, attn_axes):
    42	    """Builds einsum equations for the attention computation.
    43	
    44	    Query, key, value inputs after projection are expected to have the shape as:
    45	    `(bs, <non-attention dims>, <attention dims>, num_heads, channels)`.
    46	    `bs` and `<non-attention dims>` are treated as `<batch dims>`.
    47	
    48	    The attention operations can be generalized:
    49	    (1) Query-key dot product:
    50	    `(<batch dims>, <query attention dims>, num_heads, channels), (<batch dims>,
    51	    <key attention dims>, num_heads, channels) -> (<batch dims>,
    52	    num_heads, <query attention dims>, <key attention dims>)`
    53	    (2) Combination:
    54	    `(<batch dims>, num_heads, <query attention dims>, <key attention dims>),
    55	    (<batch dims>, <value attention dims>, num_heads, channels) -> (<batch
    56	    dims>, <query attention dims>, num_heads, channels)`
    57	
    58	    Args:
    59	      rank: Rank of query, key, value tensors.
    60	      attn_axes: List/tuple of axes, `[-1, rank)`,
    61	        that attention will be applied to.
    62	
    63	    Returns:
    64	      Einsum equations.
    65	    """
    66	    target_notation = _CHR_IDX[:rank]
    67	    # `batch_dims` includes the head dim.
    68	    batch_dims = tuple(np.delete(range(rank), attn_axes + (rank - 1,)))
    69	    letter_offset = rank
    70	    source_notation = ""
    71	    for i in range(rank):
    72	        if i in batch_dims or i == rank - 1:
    73	            source_notation += target_notation[i]
    74	        else:
    75	            source_notation += _CHR_IDX[letter_offset]
    76	            letter_offset += 1
    77	
    78	    product_notation = "".join(
    79	        [target_notation[i] for i in batch_dims]
    80	        + [target_notation[i] for i in attn_axes]
    81	        + [source_notation[i] for i in attn_axes]
    82	    )
    83	    dot_product_equation = "%s,%s->%s" % (
    84	        source_notation,
    85	        target_notation,
    86	        product_notation,
    87	    )
    88	    attn_scores_rank = len(product_notation)
    89	    combine_equation = "%s,%s->%s" % (
    90	        product_notation,
    91	        source_notation,
    92	        target_notation,
    93	    )
    94	    return dot_product_equation, combine_equation, attn_scores_rank

    98	    """Builds an einsum equation for projections inside multi-head attention."""
    99	    input_str = ""
   100	    kernel_str = ""
   101	    output_str = ""
   102	    bias_axes = ""
   103	    letter_offset = 0
   104	    for i in range(free_dims):
   105	        char = _CHR_IDX[i + letter_offset]
   106	        input_str += char
   107	        output_str += char
   108	
   109	    letter_offset += free_dims
   110	    for i in range(bound_dims):
   111	        char = _CHR_IDX[i + letter_offset]
   112	        input_str += char
   113	        kernel_str += char
   114	
   115	    letter_offset += bound_dims
   116	    for i in range(output_dims):
   117	        char = _CHR_IDX[i + letter_offset]
   118	        kernel_str += char
   119	        output_str += char
   120	        bias_axes += char
   121	    equation = "%s,%s->%s" % (input_str, kernel_str, output_str)

   130	@keras_export("keras.layers.MultiHeadAttention")
   131	class MultiHeadAttention(Layer):
   132	    """MultiHeadAttention layer.
   133	
   134	    This is an implementation of multi-headed attention as described in the
   135	    paper "Attention is all you Need" (Vaswani et al., 2017).
   136	    If `query`, `key,` `value` are the same, then
   137	    this is self-attention. Each timestep in `query` attends to the
   138	    corresponding sequence in `key`, and returns a fixed-width vector.
   139	
   140	    This layer first projects `query`, `key` and `value`. These are
   141	    (effectively) a list of tensors of length `num_attention_heads`, where the
   142	    corresponding shapes are `(batch_size, <query dimensions>, key_dim)`,
   143	    `(batch_size, <key/value dimensions>, key_dim)`,
   144	    `(batch_size, <key/value dimensions>, value_dim)`.
   145	
   146	    Then, the query and key tensors are dot-producted and scaled. These are
   147	    softmaxed to obtain attention probabilities. The value tensors are then
   148	    interpolated by these probabilities, then concatenated back to a single
   149	    tensor.
   150	
   151	    Finally, the result tensor with the last dimension as value_dim can take an
   152	    linear projection and return.
   153	
   154	    When using MultiHeadAttention inside a custom Layer, the custom Layer must
   155	    implement `build()` and call MultiHeadAttention's `_build_from_signature()`.
   156	    This enables weights to be restored correctly when the model is loaded.
   157	    TODO(b/172609172): link to documentation about calling custom build
   158	    functions when used in a custom Layer.
   159	
   160	    Examples:
   161	
   162	    Performs 1D cross-attention over two sequence inputs with an attention mask.
   163	    Returns the additional attention weights over heads.
   164	
   165	    >>> layer = MultiHeadAttention(num_heads=2, key_dim=2)
   166	    >>> target = tf.keras.Input(shape=[8, 16])
   167	    >>> source = tf.keras.Input(shape=[4, 16])
   168	    >>> output_tensor, weights = layer(target, source,
   169	    ...                                return_attention_scores=True)
   170	    >>> print(output_tensor.shape)
   171	    (None, 8, 16)
   172	    >>> print(weights.shape)
   173	    (None, 2, 8, 4)
   174	
   175	    Performs 2D self-attention over a 5D input tensor on axes 2 and 3.
   176	
   177	    >>> layer = MultiHeadAttention(
   178	    ...     num_heads=2, key_dim=2, attention_axes=(2, 3))
   179	    >>> input_tensor = tf.keras.Input(shape=[5, 3, 4, 16])
   180	    >>> output_tensor = layer(input_tensor, input_tensor)
   181	    >>> print(output_tensor.shape)
   182	    (None, 5, 3, 4, 16)
   183	
   184	    Args:
   185	      num_heads: Number of attention heads.
   186	      key_dim: Size of each attention head for query and key.
   187	      value_dim: Size of each attention head for value.
   188	      dropout: Dropout probability.
   189	      use_bias: Boolean, whether the dense layers use bias vectors/matrices.
   190	      output_shape: The expected shape of an output tensor, besides the batch
   191	        and sequence dims. If not specified, projects back to the key feature
   192	        dim.
   193	      attention_axes: axes over which the attention is applied. `None` means
   194	        attention over all axes, but batch, heads, and features.
   195	      kernel_initializer: Initializer for dense layer kernels.
   196	      bias_initializer: Initializer for dense layer biases.
   197	      kernel_regularizer: Regularizer for dense layer kernels.
   198	      bias_regularizer: Regularizer for dense layer biases.
   199	      activity_regularizer: Regularizer for dense layer activity.
   200	      kernel_constraint: Constraint for dense layer kernels.
   201	      bias_constraint: Constraint for dense layer kernels.
   202	
   203	    Call arguments:
   204	      query: Query `Tensor` of shape `(B, T, dim)`.
   205	      value: Value `Tensor` of shape `(B, S, dim)`.
   206	      key: Optional key `Tensor` of shape `(B, S, dim)`. If not given, will use
   207	        `value` for both `key` and `value`, which is the most common case.
   208	      attention_mask: a boolean mask of shape `(B, T, S)`, that prevents
   209	        attention to certain positions. The boolean mask specifies which query
   210	        elements can attend to which key elements, 1 indicates attention and 0
   211	        indicates no attention. Broadcasting can happen for the missing batch
   212	        dimensions and the head dimension.
   213	      return_attention_scores: A boolean to indicate whether the output should
   214	        be `(attention_output, attention_scores)` if `True`, or
   215	        `attention_output` if `False`. Defaults to `False`.
   216	      training: Python boolean indicating whether the layer should behave in
   217	        training mode (adding dropout) or in inference mode (no dropout).
   218	        Defaults to either using the training mode of the parent layer/model,
   219	        or False (inference) if there is no parent layer.
   220	
   221	    Returns:
   222	      attention_output: The result of the computation, of shape `(B, T, E)`,
   223	        where `T` is for target sequence shapes and `E` is the query input last
   224	        dimension if `output_shape` is `None`. Otherwise, the multi-head outputs
   225	        are project to the shape specified by `output_shape`.
   226	      attention_scores: [Optional] multi-head attention coefficients over
   227	        attention axes.

   300	    @classmethod
   301	    def from_config(cls, config):
   302	        # If the layer has a different build() function from the Keras default,
   303	        # we need to trigger the customized build to create weights.
   304	        query_shape = config.pop("query_shape")
   305	        key_shape = config.pop("key_shape")
   306	        value_shape = config.pop("value_shape")
   307	        layer = cls(**config)
   308	        if None in [query_shape, key_shape, value_shape]:
   309	            logging.warning(
   310	                "One of dimensions of the input shape is missing. It "
   311	                "should have been memorized when the layer was serialized. "
   312	                "%s is created without weights.",
   313	                str(cls),
   314	            )
   315	        else:
   316	            layer._build_from_signature(query_shape, value_shape, key_shape)
   317	        return layer

   319	    def _build_from_signature(self, query, value, key=None):
   320	        """Builds layers and variables.
   321	
   322	        Once the method is called, self._built_from_signature will be set to
   323	        True.
   324	
   325	        Args:
   326	          query: Query tensor or TensorShape.
   327	          value: Value tensor or TensorShape.
   328	          key: Key tensor or TensorShape.
   329	        """
   330	        self._built_from_signature = True
   331	        if hasattr(query, "shape"):
   332	            self._query_shape = tf.TensorShape(query.shape)
   333	        else:
   334	            self._query_shape = tf.TensorShape(query)
   335	        if hasattr(value, "shape"):
   336	            self._value_shape = tf.TensorShape(value.shape)
   337	        else:
   338	            self._value_shape = tf.TensorShape(value)
   339	        if key is None:
   340	            self._key_shape = self._value_shape
   341	        elif hasattr(key, "shape"):
   342	            self._key_shape = tf.TensorShape(key.shape)
   343	        else:
   344	            self._key_shape = tf.TensorShape(key)
   345	
   346	        # Any setup work performed only once should happen in an `init_scope`
   347	        # to avoid creating symbolic Tensors that will later pollute any eager
   348	        # operations.
   349	        with tf_utils.maybe_init_scope(self):
   350	            free_dims = self._query_shape.rank - 1
   351	            einsum_equation, bias_axes, output_rank = _build_proj_equation(
   352	                free_dims, bound_dims=1, output_dims=2
   353	            )
   354	            self._query_dense = core.EinsumDense(
   355	                einsum_equation,
   356	                output_shape=_get_output_shape(
   357	                    output_rank - 1, [self._num_heads, self._key_dim]
   358	                ),
   359	                bias_axes=bias_axes if self._use_bias else None,
   360	                name="query",
   361	                **self._get_common_kwargs_for_sublayer()
   362	            )
   363	            einsum_equation, bias_axes, output_rank = _build_proj_equation(
   364	                self._key_shape.rank - 1, bound_dims=1, output_dims=2
   365	            )
   366	            self._key_dense = core.EinsumDense(
   367	                einsum_equation,
   368	                output_shape=_get_output_shape(
   369	                    output_rank - 1, [self._num_heads, self._key_dim]
   370	                ),
   371	                bias_axes=bias_axes if self._use_bias else None,
   372	                name="key",
   373	                **self._get_common_kwargs_for_sublayer()
   374	            )
   375	            einsum_equation, bias_axes, output_rank = _build_proj_equation(
   376	                self._value_shape.rank - 1, bound_dims=1, output_dims=2
   377	            )
   378	            self._value_dense = core.EinsumDense(
   379	                einsum_equation,
   380	                output_shape=_get_output_shape(
   381	                    output_rank - 1, [self._num_heads, self._value_dim]
   382	                ),
   383	                bias_axes=bias_axes if self._use_bias else None,
   384	                name="value",
   385	                **self._get_common_kwargs_for_sublayer()
   386	            )
   387	
   388	            # Builds the attention computations for multi-head dot product
   389	            # attention.  These computations could be wrapped into the keras
   390	            # attention layer once it supports mult-head einsum computations.
   391	            self._build_attention(output_rank)
   392	            self._output_dense = self._make_output_dense(
   393	                free_dims,
   394	                self._get_common_kwargs_for_sublayer(),
   395	                "attention_output",
   396	            )

   398	    def _get_common_kwargs_for_sublayer(self):
   399	        common_kwargs = dict(
   400	            kernel_regularizer=self._kernel_regularizer,
   401	            bias_regularizer=self._bias_regularizer,
   402	            activity_regularizer=self._activity_regularizer,
   403	            kernel_constraint=self._kernel_constraint,
   404	            bias_constraint=self._bias_constraint,
   405	        )
   406	        # Create new clone of kernel/bias initializer, so that we don't reuse
   407	        # the initializer instance, which could lead to same init value since
   408	        # initializer is stateless.
   409	        kernel_initializer = self._kernel_initializer.__class__.from_config(
   410	            self._kernel_initializer.get_config()
   411	        )
   412	        bias_initializer = self._bias_initializer.__class__.from_config(
   413	            self._bias_initializer.get_config()
   414	        )
   415	        common_kwargs["kernel_initializer"] = kernel_initializer
   416	        common_kwargs["bias_initializer"] = bias_initializer
   417	        return common_kwargs

   419	    def _make_output_dense(self, free_dims, common_kwargs, name=None):
   420	        """Builds the output projection matrix.
   421	
   422	        Args:
   423	          free_dims: Number of free dimensions for einsum equation building.
   424	          common_kwargs: Common keyword arguments for einsum layer.
   425	          name: Name for the projection layer.
   426	
   427	        Returns:
   428	          Projection layer.
   429	        """
   430	        if self._output_shape:
   431	            if not isinstance(self._output_shape, collections.abc.Sized):
   432	                output_shape = [self._output_shape]
   433	            else:
   434	                output_shape = self._output_shape
   435	        else:
   436	            output_shape = [self._query_shape[-1]]
   437	        einsum_equation, bias_axes, output_rank = _build_proj_equation(
   438	            free_dims, bound_dims=2, output_dims=len(output_shape)
   439	        )
   440	        return core.EinsumDense(
   441	            einsum_equation,
   442	            output_shape=_get_output_shape(output_rank - 1, output_shape),
   443	            bias_axes=bias_axes if self._use_bias else None,
   444	            name=name,
   445	            **common_kwargs
   446	        )

   448	    def _build_attention(self, rank):
   449	        """Builds multi-head dot-product attention computations.
   450	
   451	        This function builds attributes necessary for `_compute_attention` to
   452	        costomize attention computation to replace the default dot-product
   453	        attention.
   454	
   455	        Args:
   456	          rank: the rank of query, key, value tensors.
   457	        """
   458	        if self._attention_axes is None:
   459	            self._attention_axes = tuple(range(1, rank - 2))
   460	        else:
   461	            self._attention_axes = tuple(self._attention_axes)
   462	        (
   463	            self._dot_product_equation,
   464	            self._combine_equation,
   465	            attn_scores_rank,
   466	        ) = _build_attention_equation(rank, attn_axes=self._attention_axes)
   467	        norm_axes = tuple(
   468	            range(
   469	                attn_scores_rank - len(self._attention_axes), attn_scores_rank
   470	            )
   471	        )
   472	        self._softmax = activation.Softmax(axis=norm_axes)
   473	        self._dropout_layer = regularization.Dropout(rate=self._dropout)
   474	
   475	    def _masked_softmax(self, attention_scores, attention_mask=None):
   476	        # Normalize the attention scores to probabilities.
   477	        # `attention_scores` = [B, N, T, S]
   478	        if attention_mask is not None:
   479	            # The expand dim happens starting from the `num_heads` dimension,
   480	            # (<batch_dims>, num_heads, <query_attention_dims,
   481	            # key_attention_dims>)
   482	            mask_expansion_axis = -len(self._attention_axes) * 2 - 1
   483	            for _ in range(
   484	                len(attention_scores.shape) - len(attention_mask.shape)
   485	            ):
   486	                attention_mask = tf.expand_dims(
   487	                    attention_mask, axis=mask_expansion_axis
   488	                )
   489	        return self._softmax(attention_scores, attention_mask)
   490	
   491	    def _compute_attention(
   492	        self, query, key, value, attention_mask=None, training=None
   493	    ):
   494	        """Applies Dot-product attention with query, key, value tensors.
   495	
   496	        This function defines the computation inside `call` with projected
   497	        multi-head Q, K, V inputs. Users can override this function for
   498	        customized attention implementation.
   499	
   500	        Args:
   501	          query: Projected query `Tensor` of shape `(B, T, N, key_dim)`.
   502	          key: Projected key `Tensor` of shape `(B, T, N, key_dim)`.
   503	          value: Projected value `Tensor` of shape `(B, T, N, value_dim)`.
   504	          attention_mask: a boolean mask of shape `(B, T, S)`, that prevents
   505	            attention to certain positions.
   506	          training: Python boolean indicating whether the layer should behave in
   507	            training mode (adding dropout) or in inference mode (doing nothing).
   508	
   509	        Returns:
   510	          attention_output: Multi-headed outputs of attention computation.
   511	          attention_scores: Multi-headed attention weights.
   512	        """
   513	        # Note: Applying scalar multiply at the smaller end of einsum improves
   514	        # XLA performance, but may introduce slight numeric differences in
   515	        # the Transformer attention head.
   516	        query = tf.multiply(query, 1.0 / math.sqrt(float(self._key_dim)))
   517	
   518	        # Take the dot product between "query" and "key" to get the raw
   519	        # attention scores.
   520	        attention_scores = tf.einsum(self._dot_product_equation, key, query)
   521	
   522	        attention_scores = self._masked_softmax(
   523	            attention_scores, attention_mask
   524	        )
   525	
   526	        # This is actually dropping out entire tokens to attend to, which might
   527	        # seem a bit unusual, but is taken from the original Transformer paper.
   528	        attention_scores_dropout = self._dropout_layer(
   529	            attention_scores, training=training
   530	        )
   531	
   532	        # `context_layer` = [B, T, N, H]
   533	        attention_output = tf.einsum(
   534	            self._combine_equation, attention_scores_dropout, value
   535	        )
   536	        return attention_output, attention_scores
   537	
   538	    def call(
   539	        self,
   540	        query,
   541	        value,
   542	        key=None,
   543	        attention_mask=None,
   544	        return_attention_scores=False,
   545	        training=None,
   546	    ):
   547	        if not self._built_from_signature:
   548	            self._build_from_signature(query=query, value=value, key=key)
   549	        if key is None:
   550	            key = value
   551	
   552	        query_is_ragged = isinstance(query, tf.RaggedTensor)
   553	        if query_is_ragged:
   554	            query_lengths = query.nested_row_lengths()
   555	            query = query.to_tensor()
   556	
   557	        key_is_ragged = isinstance(key, tf.RaggedTensor)
   558	        value_is_ragged = isinstance(value, tf.RaggedTensor)
   559	        if key_is_ragged and value_is_ragged:
   560	            # Ensure they have the same shape.
   561	            bounding_shape = tf.math.maximum(
   562	                key.bounding_shape(), value.bounding_shape()
   563	            )
   564	            key = key.to_tensor(shape=bounding_shape)
   565	            value = value.to_tensor(shape=bounding_shape)
   566	        elif key_is_ragged:
   567	            key = key.to_tensor(shape=tf.shape(value))
   568	        elif value_is_ragged:
   569	            value = value.to_tensor(shape=tf.shape(key))
   570	
   571	        #   N = `num_attention_heads`
   572	        #   H = `size_per_head`
   573	        # `query` = [B, T, N ,H]
   574	        query = self._query_dense(query)
   575	
   576	        # `key` = [B, S, N, H]
   577	        key = self._key_dense(key)
   578	
   579	        # `value` = [B, S, N, H]
   580	        value = self._value_dense(value)
   581	
   582	        attention_output, attention_scores = self._compute_attention(
   583	            query, key, value, attention_mask, training
   584	        )
   585	        attention_output = self._output_dense(attention_output)
   586	
   587	        if query_is_ragged:
   588	            attention_output = tf.RaggedTensor.from_tensor(
   589	                attention_output, lengths=query_lengths
   590	            )
   591	
   592	        if return_attention_scores:
   593	            return attention_output, attention_scores
   594	        return attention_output

    41	def _build_attention_equation(rank, attn_axes):
    42	    """Builds einsum equations for the attention computation.
    43	
    44	    Query, key, value inputs after projection are expected to have the shape as:
    45	    `(bs, <non-attention dims>, <attention dims>, num_heads, channels)`.
    46	    `bs` and `<non-attention dims>` are treated as `<batch dims>`.
    47	
    48	    The attention operations can be generalized:
    49	    (1) Query-key dot product:
    50	    `(<batch dims>, <query attention dims>, num_heads, channels), (<batch dims>,
    51	    <key attention dims>, num_heads, channels) -> (<batch dims>,
    52	    num_heads, <query attention dims>, <key attention dims>)`
    53	    (2) Combination:
    54	    `(<batch dims>, num_heads, <query attention dims>, <key attention dims>),
    55	    (<batch dims>, <value attention dims>, num_heads, channels) -> (<batch
    56	    dims>, <query attention dims>, num_heads, channels)`
    57	
    58	    Args:
    59	      rank: Rank of query, key, value tensors.
    60	      attn_axes: List/tuple of axes, `[-1, rank)`,
    61	        that attention will be applied to.
    62	
    63	    Returns:
    64	      Einsum equations.
    65	    """
    66	    target_notation = _CHR_IDX[:rank]
    67	    # `batch_dims` includes the head dim.
    68	    batch_dims = tuple(np.delete(range(rank), attn_axes + (rank - 1,)))
    69	    letter_offset = rank
    70	    source_notation = ""
    71	    for i in range(rank):
    72	        if i in batch_dims or i == rank - 1:
    73	            source_notation += target_notation[i]
    74	        else:
    75	            source_notation += _CHR_IDX[letter_offset]
    76	            letter_offset += 1
    77	
    78	    product_notation = "".join(
    79	        [target_notation[i] for i in batch_dims]
    80	        + [target_notation[i] for i in attn_axes]
    81	        + [source_notation[i] for i in attn_axes]
    82	    )
    83	    dot_product_equation = "%s,%s->%s" % (
    84	        source_notation,
    85	        target_notation,
    86	        product_notation,
    87	    )
    88	    attn_scores_rank = len(product_notation)
    89	    combine_equation = "%s,%s->%s" % (
    90	        product_notation,
    91	        source_notation,
    92	        target_notation,
    93	    )
    94	    return dot_product_equation, combine_equation, attn_scores_rank

   135	    paper "Attention is all you Need" (Vaswani et al., 2017).
   136	    If `query`, `key,` `value` are the same, then
   137	    this is self-attention. Each timestep in `query` attends to the
   138	    corresponding sequence in `key`, and returns a fixed-width vector.
   139	
   140	    This layer first projects `query`, `key` and `value`. These are
   141	    (effectively) a list of tensors of length `num_attention_heads`, where the
   142	    corresponding shapes are `(batch_size, <query dimensions>, key_dim)`,
   143	    `(batch_size, <key/value dimensions>, key_dim)`,
   144	    `(batch_size, <key/value dimensions>, value_dim)`.
   145	
   146	    Then, the query and key tensors are dot-producted and scaled. These are
   147	    softmaxed to obtain attention probabilities. The value tensors are then
   148	    interpolated by these probabilities, then concatenated back to a single
   149	    tensor.
   150	
   151	    Finally, the result tensor with the last dimension as value_dim can take an
   152	    linear projection and return.
   153	
   154	    When using MultiHeadAttention inside a custom Layer, the custom Layer must
   155	    implement `build()` and call MultiHeadAttention's `_build_from_signature()`.
   156	    This enables weights to be restored correctly when the model is loaded.
   157	    TODO(b/172609172): link to documentation about calling custom build
   158	    functions when used in a custom Layer.
   159	
   160	    Examples:
   161	
   162	    Performs 1D cross-attention over two sequence inputs with an attention mask.
   163	    Returns the additional attention weights over heads.
   164	
   165	    >>> layer = MultiHeadAttention(num_heads=2, key_dim=2)
   166	    >>> target = tf.keras.Input(shape=[8, 16])
   167	    >>> source = tf.keras.Input(shape=[4, 16])
   168	    >>> output_tensor, weights = layer(target, source,
   169	    ...                                return_attention_scores=True)
   170	    >>> print(output_tensor.shape)
   171	    (None, 8, 16)
   172	    >>> print(weights.shape)
   173	    (None, 2, 8, 4)
   174	
   175	    Performs 2D self-attention over a 5D input tensor on axes 2 and 3.
   176	
   177	    >>> layer = MultiHeadAttention(
   178	    ...     num_heads=2, key_dim=2, attention_axes=(2, 3))
   179	    >>> input_tensor = tf.keras.Input(shape=[5, 3, 4, 16])
   180	    >>> output_tensor = layer(input_tensor, input_tensor)
   181	    >>> print(output_tensor.shape)
   182	    (None, 5, 3, 4, 16)
   183	
   184	    Args:
   185	      num_heads: Number of attention heads.
   186	      key_dim: Size of each attention head for query and key.
   187	      value_dim: Size of each attention head for value.
   188	      dropout: Dropout probability.
   189	      use_bias: Boolean, whether the dense layers use bias vectors/matrices.
   190	      output_shape: The expected shape of an output tensor, besides the batch
   191	        and sequence dims. If not specified, projects back to the key feature
   192	        dim.
   193	      attention_axes: axes over which the attention is applied. `None` means
   194	        attention over all axes, but batch, heads, and features.
   195	      kernel_initializer: Initializer for dense layer kernels.
   196	      bias_initializer: Initializer for dense layer biases.
   197	      kernel_regularizer: Regularizer for dense layer kernels.
   198	      bias_regularizer: Regularizer for dense layer biases.
   199	      activity_regularizer: Regularizer for dense layer activity.
   200	      kernel_constraint: Constraint for dense layer kernels.
   201	      bias_constraint: Constraint for dense layer kernels.
   202	
   203	    Call arguments:
   204	      query: Query `Tensor` of shape `(B, T, dim)`.
   205	      value: Value `Tensor` of shape `(B, S, dim)`.
   206	      key: Optional key `Tensor` of shape `(B, S, dim)`. If not given, will use
   207	        `value` for both `key` and `value`, which is the most common case.
   208	      attention_mask: a boolean mask of shape `(B, T, S)`, that prevents
   209	        attention to certain positions. The boolean mask specifies which query
   210	        elements can attend to which key elements, 1 indicates attention and 0
   211	        indicates no attention. Broadcasting can happen for the missing batch
   212	        dimensions and the head dimension.
   213	      return_attention_scores: A boolean to indicate whether the output should
   214	        be `(attention_output, attention_scores)` if `True`, or
   215	        `attention_output` if `False`. Defaults to `False`.
   216	      training: Python boolean indicating whether the layer should behave in
   217	        training mode (adding dropout) or in inference mode (no dropout).
   218	        Defaults to either using the training mode of the parent layer/model,
   219	        or False (inference) if there is no parent layer.
   220	
   221	    Returns:
   222	      attention_output: The result of the computation, of shape `(B, T, E)`,
   223	        where `T` is for target sequence shapes and `E` is the query input last
   224	        dimension if `output_shape` is `None`. Otherwise, the multi-head outputs
   225	        are project to the shape specified by `output_shape`.
   226	      attention_scores: [Optional] multi-head attention coefficients over
   227	        attention axes.
   228	    """

   300	    @classmethod
   301	    def from_config(cls, config):
   302	        # If the layer has a different build() function from the Keras default,
   303	        # we need to trigger the customized build to create weights.
   304	        query_shape = config.pop("query_shape")
   305	        key_shape = config.pop("key_shape")
   306	        value_shape = config.pop("value_shape")
   307	        layer = cls(**config)
   308	        if None in [query_shape, key_shape, value_shape]:
   309	            logging.warning(
   310	                "One of dimensions of the input shape is missing. It "
   311	                "should have been memorized when the layer was serialized. "
   312	                "%s is created without weights.",
   313	                str(cls),
   314	            )
   315	        else:
   316	            layer._build_from_signature(query_shape, value_shape, key_shape)
   317	        return layer

   319	    def _build_from_signature(self, query, value, key=None):
   320	        """Builds layers and variables.
   321	
   322	        Once the method is called, self._built_from_signature will be set to
   323	        True.
   324	
   325	        Args:
   326	          query: Query tensor or TensorShape.
   327	          value: Value tensor or TensorShape.
   328	          key: Key tensor or TensorShape.
   329	        """
   330	        self._built_from_signature = True
   331	        if hasattr(query, "shape"):
   332	            self._query_shape = tf.TensorShape(query.shape)
   333	        else:
   334	            self._query_shape = tf.TensorShape(query)
   335	        if hasattr(value, "shape"):
   336	            self._value_shape = tf.TensorShape(value.shape)
   337	        else:
   338	            self._value_shape = tf.TensorShape(value)
   339	        if key is None:
   340	            self._key_shape = self._value_shape
   341	        elif hasattr(key, "shape"):
   342	            self._key_shape = tf.TensorShape(key.shape)
   343	        else:
   344	            self._key_shape = tf.TensorShape(key)
   345	
   346	        # Any setup work performed only once should happen in an `init_scope`
   347	        # to avoid creating symbolic Tensors that will later pollute any eager
   348	        # operations.
   349	        with tf_utils.maybe_init_scope(self):
   350	            free_dims = self._query_shape.rank - 1
   351	            einsum_equation, bias_axes, output_rank = _build_proj_equation(
   352	                free_dims, bound_dims=1, output_dims=2
   353	            )
   354	            self._query_dense = core.EinsumDense(
   355	                einsum_equation,
   356	                output_shape=_get_output_shape(
   357	                    output_rank - 1, [self._num_heads, self._key_dim]
   358	                ),
   359	                bias_axes=bias_axes if self._use_bias else None,
   360	                name="query",
   361	                **self._get_common_kwargs_for_sublayer()
   362	            )
   363	            einsum_equation, bias_axes, output_rank = _build_proj_equation(
   364	                self._key_shape.rank - 1, bound_dims=1, output_dims=2
   365	            )
   366	            self._key_dense = core.EinsumDense(
   367	                einsum_equation,
   368	                output_shape=_get_output_shape(
   369	                    output_rank - 1, [self._num_heads, self._key_dim]
   370	                ),
   371	                bias_axes=bias_axes if self._use_bias else None,
   372	                name="key",
   373	                **self._get_common_kwargs_for_sublayer()
   374	            )
   375	            einsum_equation, bias_axes, output_rank = _build_proj_equation(
   376	                self._value_shape.rank - 1, bound_dims=1, output_dims=2
   377	            )
   378	            self._value_dense = core.EinsumDense(
   379	                einsum_equation,
   380	                output_shape=_get_output_shape(
   381	                    output_rank - 1, [self._num_heads, self._value_dim]
   382	                ),
   383	                bias_axes=bias_axes if self._use_bias else None,
   384	                name="value",
   385	                **self._get_common_kwargs_for_sublayer()
   386	            )
   387	
   388	            # Builds the attention computations for multi-head dot product
   389	            # attention.  These computations could be wrapped into the keras
   390	            # attention layer once it supports mult-head einsum computations.
   391	            self._build_attention(output_rank)
   392	            self._output_dense = self._make_output_dense(
   393	                free_dims,
   394	                self._get_common_kwargs_for_sublayer(),
   395	                "attention_output",
   396	            )

   398	    def _get_common_kwargs_for_sublayer(self):
   399	        common_kwargs = dict(
   400	            kernel_regularizer=self._kernel_regularizer,
   401	            bias_regularizer=self._bias_regularizer,
   402	            activity_regularizer=self._activity_regularizer,
   403	            kernel_constraint=self._kernel_constraint,
   404	            bias_constraint=self._bias_constraint,
   405	        )
   406	        # Create new clone of kernel/bias initializer, so that we don't reuse
   407	        # the initializer instance, which could lead to same init value since
   408	        # initializer is stateless.
   409	        kernel_initializer = self._kernel_initializer.__class__.from_config(
   410	            self._kernel_initializer.get_config()
   411	        )
   412	        bias_initializer = self._bias_initializer.__class__.from_config(
   413	            self._bias_initializer.get_config()
   414	        )
   415	        common_kwargs["kernel_initializer"] = kernel_initializer
   416	        common_kwargs["bias_initializer"] = bias_initializer
   417	        return common_kwargs

   419	    def _make_output_dense(self, free_dims, common_kwargs, name=None):
   420	        """Builds the output projection matrix.
   421	
   422	        Args:
   423	          free_dims: Number of free dimensions for einsum equation building.
   424	          common_kwargs: Common keyword arguments for einsum layer.
   425	          name: Name for the projection layer.
   426	
   427	        Returns:
   428	          Projection layer.
   429	        """
   430	        if self._output_shape:
   431	            if not isinstance(self._output_shape, collections.abc.Sized):
   432	                output_shape = [self._output_shape]
   433	            else:
   434	                output_shape = self._output_shape
   435	        else:
   436	            output_shape = [self._query_shape[-1]]
   437	        einsum_equation, bias_axes, output_rank = _build_proj_equation(
   438	            free_dims, bound_dims=2, output_dims=len(output_shape)
   439	        )
   440	        return core.EinsumDense(
   441	            einsum_equation,
   442	            output_shape=_get_output_shape(output_rank - 1, output_shape),
   443	            bias_axes=bias_axes if self._use_bias else None,
   444	            name=name,
   445	            **common_kwargs
   446	        )

   448	    def _build_attention(self, rank):
   449	        """Builds multi-head dot-product attention computations.
   450	
   451	        This function builds attributes necessary for `_compute_attention` to
   452	        costomize attention computation to replace the default dot-product
   453	        attention.
   454	
   455	        Args:
   456	          rank: the rank of query, key, value tensors.
   457	        """
   458	        if self._attention_axes is None:
   459	            self._attention_axes = tuple(range(1, rank - 2))
   460	        else:
   461	            self._attention_axes = tuple(self._attention_axes)
   462	        (
   463	            self._dot_product_equation,
   464	            self._combine_equation,
   465	            attn_scores_rank,
   466	        ) = _build_attention_equation(rank, attn_axes=self._attention_axes)
   467	        norm_axes = tuple(
   468	            range(
   469	                attn_scores_rank - len(self._attention_axes), attn_scores_rank
   470	            )
   471	        )
   472	        self._softmax = activation.Softmax(axis=norm_axes)
   473	        self._dropout_layer = regularization.Dropout(rate=self._dropout)
   474	
   475	    def _masked_softmax(self, attention_scores, attention_mask=None):
   476	        # Normalize the attention scores to probabilities.
   477	        # `attention_scores` = [B, N, T, S]
   478	        if attention_mask is not None:
   479	            # The expand dim happens starting from the `num_heads` dimension,
   480	            # (<batch_dims>, num_heads, <query_attention_dims,
   481	            # key_attention_dims>)
   482	            mask_expansion_axis = -len(self._attention_axes) * 2 - 1
   483	            for _ in range(
   484	                len(attention_scores.shape) - len(attention_mask.shape)
   485	            ):
   486	                attention_mask = tf.expand_dims(
   487	                    attention_mask, axis=mask_expansion_axis
   488	                )
   489	        return self._softmax(attention_scores, attention_mask)
   490	
   491	    def _compute_attention(
   492	        self, query, key, value, attention_mask=None, training=None
   493	    ):
   494	        """Applies Dot-product attention with query, key, value tensors.
   495	
   496	        This function defines the computation inside `call` with projected
   497	        multi-head Q, K, V inputs. Users can override this function for
   498	        customized attention implementation.
   499	
   500	        Args:
   501	          query: Projected query `Tensor` of shape `(B, T, N, key_dim)`.
   502	          key: Projected key `Tensor` of shape `(B, T, N, key_dim)`.
   503	          value: Projected value `Tensor` of shape `(B, T, N, value_dim)`.
   504	          attention_mask: a boolean mask of shape `(B, T, S)`, that prevents
   505	            attention to certain positions.
   506	          training: Python boolean indicating whether the layer should behave in
   507	            training mode (adding dropout) or in inference mode (doing nothing).
   508	
   509	        Returns:
   510	          attention_output: Multi-headed outputs of attention computation.
   511	          attention_scores: Multi-headed attention weights.
   512	        """
   513	        # Note: Applying scalar multiply at the smaller end of einsum improves
   514	        # XLA performance, but may introduce slight numeric differences in
   515	        # the Transformer attention head.
   516	        query = tf.multiply(query, 1.0 / math.sqrt(float(self._key_dim)))
   517	
   518	        # Take the dot product between "query" and "key" to get the raw
   519	        # attention scores.
   520	        attention_scores = tf.einsum(self._dot_product_equation, key, query)
   521	
   522	        attention_scores = self._masked_softmax(
   523	            attention_scores, attention_mask
   524	        )
   525	
   526	        # This is actually dropping out entire tokens to attend to, which might
   527	        # seem a bit unusual, but is taken from the original Transformer paper.
   528	        attention_scores_dropout = self._dropout_layer(
   529	            attention_scores, training=training
   530	        )
   531	
   532	        # `context_layer` = [B, T, N, H]
   533	        attention_output = tf.einsum(
   534	            self._combine_equation, attention_scores_dropout, value
   535	        )
   536	        return attention_output, attention_scores
   537	
   538	    def call(
   539	        self,
   540	        query,
   541	        value,
   542	        key=None,
   543	        attention_mask=None,
   544	        return_attention_scores=False,
   545	        training=None,
   546	    ):
   547	        if not self._built_from_signature:
   548	            self._build_from_signature(query=query, value=value, key=key)
   549	        if key is None:
   550	            key = value
   551	
   552	        query_is_ragged = isinstance(query, tf.RaggedTensor)
   553	        if query_is_ragged:
   554	            query_lengths = query.nested_row_lengths()
   555	            query = query.to_tensor()
   556	
   557	        key_is_ragged = isinstance(key, tf.RaggedTensor)
   558	        value_is_ragged = isinstance(value, tf.RaggedTensor)
   559	        if key_is_ragged and value_is_ragged:
   560	            # Ensure they have the same shape.
   561	            bounding_shape = tf.math.maximum(
   562	                key.bounding_shape(), value.bounding_shape()
   563	            )
   564	            key = key.to_tensor(shape=bounding_shape)
   565	            value = value.to_tensor(shape=bounding_shape)
   566	        elif key_is_ragged:
   567	            key = key.to_tensor(shape=tf.shape(value))
   568	        elif value_is_ragged:
   569	            value = value.to_tensor(shape=tf.shape(key))
   570	
   571	        #   N = `num_attention_heads`
   572	        #   H = `size_per_head`
   573	        # `query` = [B, T, N ,H]
   574	        query = self._query_dense(query)
   575	
   576	        # `key` = [B, S, N, H]
   577	        key = self._key_dense(key)
   578	
   579	        # `value` = [B, S, N, H]
   580	        value = self._value_dense(value)
   581	
   582	        attention_output, attention_scores = self._compute_attention(
   583	            query, key, value, attention_mask, training
   584	        )
   585	        attention_output = self._output_dense(attention_output)
   586	
   587	        if query_is_ragged:
   588	            attention_output = tf.RaggedTensor.from_tensor(
   589	                attention_output, lengths=query_lengths
   590	            )
   591	
   592	        if return_attention_scores:
   593	            return attention_output, attention_scores
   594	        return attention_output

    15	def get_layer_and_inputs(num_heads=2, key_dim=4, value_dim=8, batch_size=1, query_len=3, value_len=3, dtype=tf.float32):
    16	    """Helper to create a MultiHeadAttention layer and valid inputs."""
    17	    layer = MultiHeadAttention(num_heads=num_heads, key_dim=key_dim, value_dim=value_dim)
    18	    # Build the layer to initialize weights
    19	    dummy_q = tf.zeros((batch_size, 1, num_heads, key_dim))
    20	    dummy_k = tf.zeros((batch_size, 1, num_heads, key_dim))
    21	    dummy_v = tf.zeros((batch_size, 1, num_heads, value_dim))
    22	    layer(dummy_q, dummy_k, dummy_v)
    23	    
    24	    # Create actual inputs
    25	    # Shape: (B, T, N, key_dim) for Q and K
    26	    query = tf.random.normal((batch_size, query_len, num_heads, key_dim), dtype=dtype)
    27	    key = tf.random.normal((batch_size, value_len, num_heads, key_dim), dtype=dtype)
    28	    value = tf.random.normal((batch_size, value_len, num_heads, value_dim), dtype=dtype)
    29	    
    30	    return layer, query, key, value
